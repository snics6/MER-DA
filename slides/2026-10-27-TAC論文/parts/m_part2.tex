\begin{frame}{同じモダリティの先生でも生徒は上がるが，その先生の $J$ は小さい}

      \vspace{0.3em}
      {\raggedright\tiny $J$ ＝（先生だけが当てる割合）−（出発点だけが当てる割合）を感情ごとに平均したもの ＝ プールでの先生の UAR − 出発点の UAR\par}
    \end{column}
    \begin{column}{0.54\textwidth}
      \begin{itemize}\scriptsize\setlength{\itemsep}{5pt}
        \item 出発点に縛った生徒は先生の優位をおおよそ再現するのが精一杯なので，生徒の利得のうち先生の情報から来る分は $J$ を超えないと予想する（定理ではない）
        \item 見込み：ソースごとの分類器を，仮のターゲットでの混同の仕方で重みづけてまとめた先生．A では $J$ = \NUM{p2-e52-q-minus-p-a} \pt
        \item その弱いラベルで，A は $K = 0, 1, 5$ に \NUM{p2-e59k-a-k0}，\NUM{p2-e59k-a-k1}，\NUM{p2-e59k-a-k5} \pt，V は \NUM{p2-e59k-v-k0}，\NUM{p2-e59k-v-k1}，\NUM{p2-e59k-v-k5} \pt 上がる（反復 10 回）．$K=25$ までに消える
        \item A の $K=0$ では，出発点の予測の釣り合い直しだけで \NUM{p2-e59k-ctrl-a-k0} \pt．見込みの情報の分は $K\le5$ で \NUM{p2-e59k-info-a-lo}〜\NUM{p2-e59k-info-a-hi} \pt で，$J$ より小さい．言葉ではほぼ 0
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{拡張版は，$K$ 本で学び直した先生を候補に足して，$K\ge10$ の損を消す}
  \rl{手法：基本版と拡張版}
  \vspace{0.3em}
  \centering\scriptsize
  \renewcommand{\arraystretch}{1.35}
  \begin{tabularx}{\textwidth}{@{}l X X@{}}
    \toprule
    & \textbf{基本版} & \textbf{拡張版}（基本版の候補に足すもの） \\
    \midrule
    単独の生徒の先生 & 融合の出発点（ソースだけで学ぶ） & $K$ 本で学び直した融合の生徒 \\
    融合の生徒の先生 & 融合の出発点が教えた単独の生徒の予測の積 & $K$ 本で学び直した融合の生徒が教えた単独の生徒の予測の積 \\
    弱いラベルの重みづけ & 感情ごとの合計をそろえる & そろえない，しきい値を超えた発話だけを硬いラベルにする \\
    $K$ ごとに選ぶもの & 先生，$m$ & 先生，$m$，重みづけ \\
    \bottomrule
  \end{tabularx}\par
  \vspace{0.6em}\raggedright
  \begin{itemize}\scriptsize\setlength{\itemsep}{4pt}
    \item 拡張版は，比較で評価する AVT の音声の生徒と AV の融合の生徒で，基本版の $K\ge10$ の損を消した（VT の言葉と AT の音声の生徒は拡張版を流していない）
    \item 基本版の弱いラベルはすべてソースだけで学んだ融合の出発点から作るが，融合の生徒の先生はターゲットのラベル無しの発話を使う．「基本版の先生はターゲットのデータを見ない」とは言えない
  \end{itemize}
\end{frame}

\begin{frame}{第 II 部のまとめ：向きの情報は，別のモダリティの弱いラベルから得る}
  \rl{まとめ}
  \vspace{0.3em}
  \begin{itemize}\footnotesize\setlength{\itemsep}{7pt}
    \item 出発点：中心化，全ソースからの仮説転移，5 つの値の選択の順に UAR が上がり，比べた作り方の中で最も高い出発点になった
    \item 少ないラベルが直せる向きは $4K$ 本の張る空間の中だけで（命題 1），ラベルを見ない選び方ではそこを向きの誤差に合わせられない（命題 2，実測でもでたらめの \NUM{p2-e58-ratio-unlab-lo}〜\NUM{p2-e58-ratio-unlab-hi} 倍）
    \item プール全体への弱いラベルは，数千本の発話にラベルの情報を足す．先生の情報から来る利得は先生の優位 $J$ を超えないと予想し（定理ではない），同じモダリティの先生では $J$ が小さい
    \item ソースだけで学んだ融合の出発点を先生にすると，$K\le5$ ですべての生徒が出発点を上回る．$K\ge10$ の損は拡張版の候補で消える
  \end{itemize}
\end{frame}
